17. Create Instructor collection and execute following queries
 a. Find Instructor with id 1,2,3
 b. Find details of instructor except 1,2,3
 c. Find instructor whose salary is neither < 50000 nor department is computer
 d. Display details of instructor dept has substring e

use mongo16; 

db.Instructor.insertMany([
{ _id: 1, first_name: "Saurabh", last_name: "Kale", dept: "IT", salary: 45000, phone: ["9876543210"], status: "approved" },
{ _id: 2, first_name: "Rohit", last_name: "Patil", dept: "COMP", salary: 39000, phone: ["9988776655"], status: "pending" },
{ _id: 3, first_name: "Priya", last_name: "Deshmukh", dept: "ENTC", salary: 52000, phone: ["9090909090"], status: "approved" },
{ _id: 4, first_name: "Amit", last_name: "Joshi", dept: "IT", salary: 48000, phone: ["9123456789"], status: "approved" },
{ _id: 5, first_name: "Sneha", last_name: "Kulkarni", dept: "COMP", salary: 41000, phone: ["9876501234"], status: "pending" }
])

// a. Find Instructor with id 1,2,3 

db.Instructor.find({ _id: { $in: [1, 2, 3] } }); 

// b. Find details of instructor except 1,2,3 

db.Instructor.find({ _id: { $nin: [1, 2, 3] } }) 

// c. Find instructor whose salary is neither < 50000 nor department is computer 
 
db.Instructor.find({$nor:[{salary:{$lt:50000}},{dept:"COMP"}]}) 

// d. Display details of instructor dept has substring e 

db.Instructor.find({ dept: { $regex: /e/i } })
