// Display all teacher details 

db.teachers.find().pretty();

// Find teachers from IT department 

db.teachers.find({dept:"it"})

// Find teachers whose salary is greater than ₹60,000 

db.teachers.find({salary:{$gt:60000}})

// Find teachers whose status is “approved” 

db.teachers.find({status:"approved"}); 

// Display teachers in ascending order of salary 

db.teachers.find().sort({ salary: 1 }) 

// Display teachers in descending order of salary 

db.teachers.find().sort({salary:-1})  

// Find the total salary of all approved teachers 
 
db.teachers.aggregate([{$match:{status:"approved"}},{$group:{_id:null,total_salary:{$sum:"$salary"}}}]) 

// Increment salary of teachers in IT dept by ₹5,000
 
db.teachers.updateMany({dept:"it"},{$inc:{salary:5000}});

// Remove teachers whose status is not approved 
 
db.teachers.deleteMany({ status: " not approved" });

// Count the number of teachers in each department 

db.teachers.aggregate([ { $group: { _id: "$dept", total_teachers: { $sum: 1 } } } ])
