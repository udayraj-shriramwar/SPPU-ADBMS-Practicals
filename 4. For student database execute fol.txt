//  Find the record of the students who has got the highest marks in DBMS subject.

db.records.find().sort({dbms:-1}).limit(1);

// Find the average result of TOC subject. 

db.records.aggregate([{ $group: {_id:null,avg_toc:{ $avg:"$toc"}}}]); 

// Find the record of the students who has got the lowest marks in CNT subject.

db.records.find().sort({cnt:1}).limit(1);

// Find the total number of students who scored first class.

db.records.aggregate([{$match:{class:"Distinction"}},{$count:"total_dis_stu"}]);
