 db.employees.insertOne({emp_id:1,name:"rahul",dept:"hr" ,salary:45000,status:"active"});

// Using Modifiers ($set, $inc, $rename) 

db.employees.updateOne({ emp_id: 2 }, { $set: { department: "R&D" } })

db.employees.updateOne({ emp_id: 3 }, { $inc: { salary: 5000 } })

db.employees.updateMany({}, { $rename: { "department": "dept" } })

// 3. Upsert (Insert if Not Found) 

db.employees.updateOne({ emp_id: 6 }, { $set: { name: "Rohit Pawar", dept: "HR", salary: 42000 } }, { upsert: true })


// 4. Update Multiple Documents 

db.employees.updateMany({ dept: "IT" }, { $inc: { salary: 2000 } })
 
// 5. Returning Updated Document (Mongo Shell)
